        async function startBattle() {
            if (!requireWallet()) return;
            if (!battleOrder || battleOrder.length < 3) {
                showToast('⚠️ 请先设置出战顺序（点「调整」勾选 3 张）', 'error'); return;
            }
            const diff = Number(document.querySelector('.difficulty-selector .btn-sm.active')?.getAttribute('data-diff') || 1);
            const fee = [10000, 15000, 20000, 25000][diff-1] || 10000;
            if (depositBalance < fee*1_000_000) { showToast('❌ 存款不足', 'error'); return; }
            depositBalance -= fee * 1_000_000;
            mock_ai_today++;
            const aiPool = [mockDrawOne(), mockDrawOne(), mockDrawOne()];
            const myCards = battleOrder.map(id => userCards.find(c => c.instance_id === id)).filter(Boolean);
            if (myCards.length < 3) { showToast('⚠️ 出战卡牌不存在', 'error'); return; }
            const diffBoost = [1.0, 1.0, 1.15, 1.30, 1.50][diff] || 1.0;
            let wins=0, losses=0;
            for (let i=0;i<3;i++) {
                const p = myCards[i], a = aiPool[i];
                const pp = p.attack + p.defense;
                const ap = Math.round((a.attack + a.defense) * diffBoost);
                if (pp >= ap) wins++; else losses++;
            }
            if (wins >= 2) {
                const reward = [40000, 50000, 70000, 80000][diff-1] * 1_000_000;
                depositBalance += reward;
                showToast(`🏆 胜利！${wins}-${losses} · +${(reward/1e6).toFixed(0)} TKCC`, 'success');
            } else {
                showToast(`💀 失败 ${wins}-${losses}`, 'info');
            }
            refreshBalances();
            refreshAiStatsAndUI();
            try { displayBattleResult({ win: wins>=2, rounds: myCards.map((c,i)=>({me:c.name,ai:aiPool[i].name,win:(myCards[i].attack+myCards[i].defense)>=Math.round((aiPool[i].attack+aiPool[i].defense)*diffBoost)})) }, myCards, aiPool); } catch(e){}
        }

        function cancelFromList(matchId) {

        async function doAcceptPvpMatch() {
            if (!_currentPvpMatchId) return;
            const idx = mockRooms.findIndex(r => r.match_id === _currentPvpMatchId);
            if (idx === -1) { showToast('❌ 房间不存在', 'error'); return; }
            const room = mockRooms[idx];
            if (room.status !== 'waiting') { showToast('⚠️ 房间已结束', 'error'); return; }
            if (_selectedCards.length !== 3) { showToast('⚠️ 请选 3 张卡', 'error'); return; }
            if (depositBalance < 60000_000000) { showToast('❌ 存款不足', 'error'); return; }
            depositBalance -= 60000_000000;
            // 模拟 3 局 2 胜结算
            room.opponent = walletAddress;
            room.opponent_order = _selectedCards.slice();
            const myCards = _selectedCards.map(id => userCards.find(c=>c.instance_id===id)).filter(Boolean);
            // 用 challenger_order 从 userCards 找对手卡
            const oppCards = room.challenger_order.map(id => userCards.find(c=>c.instance_id===id)).filter(Boolean);
            let wins = 0;
            for (let i=0;i<3;i++) {
                const mp=(myCards[i]?.attack||0)+(myCards[i]?.defense||0);
                const op=(oppCards[i]?.attack||0)+(oppCards[i]?.defense||0);
                if (mp+Math.floor(Math.random()*30) >= op) wins++;
            }
            const win = wins >= 2;
            room.winner = win ? walletAddress : room.challenger;
            room.status = 'finished';
            room.finished_at = Math.floor(Date.now() / 1000);
            if (win) {
                const reward = 2*60000_000000 + 100000_000000;
                depositBalance += reward;
                refreshBalances();
                showToast(`🏆 你赢了 ${wins}-${3-wins}！+${(reward/1e6).toFixed(0)} TKCC`, 'success');
            } else {
                refreshBalances();
                showToast(`💀 你输了 ${wins}-${3-wins}`, 'info');
            }
            _currentPvpMatchId = null;
            closePvpAcceptModal();
            listPvpUI();
        }

        async function doCreateRoyale() {
            if (!requireWallet()) return;
            if (_selectedCards.length !== 3) { showToast('⚠️ 请选 3 张卡', 'error'); return; }
            const size = Number(document.getElementById('royaleSize').value);

            depositBalance -= 60000_000000;
            showToast(`👑 创建 ${size} 人混战房间！本地 AI 挤满中...`, 'success');
            closeRoyaleCreateModal();
            refreshBalances();
            setTimeout(() => {
                const allOrders = [_selectedCards.map(id=>userCards.find(c=>c.instance_id===id)).filter(Boolean)];
                for (let i=1;i<size;i++) allOrders.push(userCards.sort(()=>Math.random()-0.5).slice(0,3));
                const scores = new Array(size).fill(0);
                for (let i=0;i<size;i++) for (let j=i+1;j<size;j++) {
                    let wi=0;
                    for (let k=0;k<3;k++) {
                        const pi=allOrders[i][k], pj=allOrders[j][k];
                        if (!pi||!pj) continue;
                        const pp=pi.attack+pi.defense, ap=pj.attack+pj.defense;
                        if (pp>ap) wi++; else if (ap>pp) wi--;
                    }
                    scores[i]+=wi; scores[j]-=wi;
                }
                const winnerIdx = scores.indexOf(Math.max(...scores));
                const pool = size*60000_000000;
                if (winnerIdx===0) { depositBalance += Math.floor(pool*0.70); }
                refreshBalances();
                showToast(winnerIdx===0 ? `🏆 混战夺冠 +${(pool*0.70/1e6).toFixed(0)}` : `💀 混战第 ${winnerIdx+1} 名`, winnerIdx===0?'success':'info');
            }, 1200);
        }

        async function confirmUpgrade() {
            if (!detailCard) return;
            const fees = [50000, 150000, 400000, 1000000];
            const fee = fees[(detailCard.star||1)-1] * 1_000_000;
            if (depositBalance < fee) { showToast('❌ 存款不足', 'error'); return; }
            if ((detailCard.star||1) >= 5) { showToast('⚠️ 已达满星', 'error'); return; }
            depositBalance -= fee;
            detailCard.star = (detailCard.star||1) + 1;
            refreshBalances();
            showToast(`⭐ ${detailCard.name} 升1星！现在 ${detailCard.star} 星`, 'success');
            showCardDetail(detailCard.instance_id);
        }
        async function confirmUpgradeToMax() {
            if (!detailCard) return;
            const fees = [50000, 150000, 400000, 1000000];
            let totalFee = 0;
            while ((detailCard.star||1) < 5) {
                totalFee += fees[(detailCard.star||1)-1] * 1_000_000;
                detailCard.star = (detailCard.star||1) + 1;
            }
            if (depositBalance < totalFee) { showToast('❌ 存款不足', 'error'); return; }
            depositBalance -= totalFee;
            refreshBalances();
            showToast(`⭐ ${detailCard.name} 升到满星！花费 ${(totalFee/1e6).toFixed(0)}`, 'success');
            showCardDetail(detailCard.instance_id);
        }
        async function confirmUpgradeFragments() {
            if (!detailCard) return;
            const fees = [50000, 150000, 400000, 1000000];
            const fragCost = Math.ceil(fees[(detailCard.star||1)-1] / 100);
            const r = detailCard.rarity;
            const have = fragmentsData[r] || 0;
            if (have < fragCost) { showToast(`❌ ${r}碎片不足（需要 ${fragCost}，有 ${have}）`, 'error'); return; }
            fragmentsData[r] = have - fragCost;
            detailCard.star = (detailCard.star||1) + 1;
            refreshBalances();
            showToast(`✨ ${detailCard.name} 碎片升星！消耗 ${fragCost} ${r}碎片`, 'success');
            showCardDetail(detailCard.instance_id);
        }
        async function decomposeCard() {
            if (!detailCard) return;
            const r = detailCard.rarity;
            const gain = { common: 5, rare: 10, epic: 20, legend: 50 }[r] || 5;
            fragmentsData[r] = (fragmentsData[r]||0) + gain;
            userCards = userCards.filter(c => c.instance_id !== detailCard.instance_id);
            const name = detailCard.name;
            detailCard = null;
            document.getElementById('cardDetailModal').style.display = 'none';
            refreshBalances();
            showToast(`🔥 ${name} 分解 +${gain} ${r}碎片`, 'success');
        }
        async function upgradeCard() {
            if (!detailCard) return;
            const r = detailCard.rarity;
            const cost = { common: 50, rare: 100, epic: 200, legend: 500 }[r] || 50;
            if ((detailCard.level||0) >= 10) { showToast('⚠️ 已满级', 'error'); return; }
            if ((fragmentsData[r]||0) < cost) { showToast(`❌ ${r}碎片不足`, 'error'); return; }
            fragmentsData[r] -= cost;
            detailCard.level = (detailCard.level||0) + 1;
            detailCard.attack += 3;
            detailCard.defense += 2;
            refreshBalances();
            showToast(`💠 ${detailCard.name} 升到 Lv.${detailCard.level} (+3攻 +2防)`, 'success');
            showCardDetail(detailCard.instance_id);
        }

            const pool = CARD_TEMPLATES.filter(c => c.rarity === rarity);

            userCards.push(nc);
            refreshBalances();
            showToast(`✨ 合成 ${nc.name}（${rarity}）成功！`, 'success');
        }
